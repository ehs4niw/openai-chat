if (!function_exists('bs_edu_staff_roles')) {
    function bs_edu_staff_roles() {
        $roles = array('administrator', 'bs_study_support');
        if (defined('BSC_ROLE_SLUGS')) $roles = array_merge($roles, (array)BSC_ROLE_SLUGS);
        if (function_exists('bsc_role_slugs')) $roles = array_merge($roles, bsc_role_slugs());
        foreach (wp_roles()->get_names() as $slug => $label) {
            $label = strtr((string)translate_user_role($label), array('ي'=>'ی', 'ك'=>'ک', "\u{200c}"=>'', ' '=>''));
            if (strpos($label, 'مدرسراهنما') !== false) $roles[] = $slug;
        }
        return array_values(array_unique($roles));
    }
}
if (!function_exists('bs_edu_is_staff')) {
    function bs_edu_is_staff($user_id = 0) {
        $user = get_userdata($user_id ? (int)$user_id : get_current_user_id());
        if (!$user) return false;
        if (user_can($user, 'manage_options') || user_can($user, 'bs_counselor')) return true;
        return (bool)array_intersect((array)$user->roles, bs_edu_staff_roles());
    }
}

if (!function_exists('bs_grades')) {
    function bs_grades() {
        return array('10'=>'دهم', '11'=>'یازدهم', '12'=>'دوازدهم');
    }
}

add_action('woocommerce_edit_account_form', 'bs_field_group');
function bs_field_group() {
    $uid = get_current_user_id();
    if (bs_edu_is_staff($uid)) return;
    $cur = get_user_meta($uid, 'edu_field', true);
    $grade = get_user_meta($uid, 'edu_grade', true);
    $opts = array('math'=>'ریاضی و فیزیک', 'science'=>'علوم تجربی', 'humanities'=>'علوم انسانی');
    echo '<p class="woocommerce-form-row form-row form-row-wide"><label for="edu_grade">پایه تحصیلی</label><select name="edu_grade" id="edu_grade" class="woocommerce-Input input-text"><option value="">انتخاب کنید</option>';
    foreach (bs_grades() as $key=>$label) echo '<option value="'.esc_attr($key).'"'.selected((string)$grade,(string)$key,false).'>'.esc_html($label).'</option>';
    echo '</select></p><p class="woocommerce-form-row form-row form-row-wide"><label for="edu_field">گروه آزمایشی</label><select name="edu_field" id="edu_field" class="woocommerce-Input input-text"><option value="">انتخاب کنید</option>';
    foreach ($opts as $key=>$label) echo '<option value="'.esc_attr($key).'"'.selected($cur,$key,false).'>'.esc_html($label).'</option>';
    echo '</select></p>';
}

add_action('woocommerce_save_account_details', 'bs_save_extra_fields');
function bs_save_extra_fields($user_id) {
    // WooCommerce verifies the account form and nonce before this hook runs.
    if (!bs_edu_is_staff($user_id)) {
        if (isset($_POST['edu_grade']) && is_scalar($_POST['edu_grade'])) {
            $grade = sanitize_text_field(wp_unslash($_POST['edu_grade']));
            if (isset(bs_grades()[$grade])) update_user_meta($user_id, 'edu_grade', $grade);
        }
        if (isset($_POST['edu_field']) && is_scalar($_POST['edu_field'])) {
            $group = sanitize_text_field(wp_unslash($_POST['edu_field']));
            if (in_array($group, array('math','science','humanities',''), true)) update_user_meta($user_id, 'edu_field', $group);
        }
    }
    // Location fields remain usable for students and staff alike.
    if (isset($_POST['bs_state']) && is_scalar($_POST['bs_state'])) {
        update_user_meta($user_id, 'billing_state', sanitize_text_field(wp_unslash($_POST['bs_state'])));
        update_user_meta($user_id, 'billing_country', 'IR');
    }
    if (isset($_POST['bs_city']) && is_scalar($_POST['bs_city'])) {
        update_user_meta($user_id, 'billing_city', sanitize_text_field(wp_unslash($_POST['bs_city'])));
    }
}

/* ===== بخش: حذف فیلدهای رمز از جزئیات حساب ===== */
add_action( 'template_redirect', 'bs_strip_password_fields', 1 );
function bs_strip_password_fields() {
    if ( ! function_exists( 'is_account_page' ) || ! is_account_page() ) {
        return;
    }
    if ( empty( $_POST['action'] ) || 'save_account_details' !== $_POST['action'] ) {
        return;
    }
    // فقط اگه هر سه فیلد رمز تکمیل نشده، همه رو پاک کن
    $cur = isset( $_POST['password_current'] ) ? $_POST['password_current'] : '';
    $n1  = isset( $_POST['password_1'] ) ? $_POST['password_1'] : '';
    $n2  = isset( $_POST['password_2'] ) ? $_POST['password_2'] : '';
    if ( '' === $n1 && '' === $n2 ) {
        unset( $_POST['password_current'], $_POST['password_1'], $_POST['password_2'] );
        unset( $_REQUEST['password_current'], $_REQUEST['password_1'], $_REQUEST['password_2'] );
    }
}

function bs_account_address_fields() {
    static $printed = false;
    if ( $printed ) { return; }
    $printed = true;

    $uid    = get_current_user_id();
    $states = WC()->countries->get_states( 'IR' );
    $states = is_array( $states ) ? $states : array();
    $cur_st = get_user_meta( $uid, 'billing_state', true );

    echo '<h4 class="bs-addr-title" style="margin:20px 0 12px;font-weight:700;">نشانی</h4>';

    echo '<p class="woocommerce-form-row form-row form-row-wide">';
    echo '<label for="bs_addr_state">استان</label>';
    echo '<select name="bs_addr_state" id="bs_addr_state" class="woocommerce-Input input-text"' . $w . '>';
    echo '<option value="">انتخاب کنید</option>';
    foreach ( $states as $code => $name ) {
        echo '<option value="' . esc_attr( $code ) . '"' . selected( (string) $cur_st, (string) $code, false ) . '>' . esc_html( $name ) . '</option>';

    echo '<label for="bs_addr_city">شهر</label>';
    echo '<input type="text" class="woocommerce-Input input-text" name="bs_addr_city" id="bs_addr_city"' . $w . ' value="' . esc_attr( get_user_meta( $uid, 'billing_city', true ) ) . '">';

    echo '<p class="woocommerce-form-row form-row form-row-wide">';
    echo '<label for="bs_addr_address">نشانی کامل</label>';
    echo '<textarea class="woocommerce-Input input-text" name="bs_addr_address" id="bs_addr_address" rows="3"' . $w . '>' . esc_textarea( get_user_meta( $uid, 'billing_address_1', true ) ) . '</textarea>';

// ۴) ذخیره‌ی نشانی (در فیلدهای صورتحساب ووکامرس)
add_action( 'woocommerce_save_account_details', 'bs_save_account_address', 20 );
function bs_save_account_address( $user_id ) {
    if ( ! isset( $_POST['bs_addr_state'] ) && ! isset( $_POST['bs_addr_city'] ) ) { return; }

    $states = WC()->countries->get_states( 'IR' );
    $state  = isset( $_POST['bs_addr_state'] ) ? sanitize_text_field( wp_unslash( $_POST['bs_addr_state'] ) ) : '';
    if ( '' === $state || ( is_array( $states ) && isset( $states[ $state ] ) ) ) {
        update_user_meta( $user_id, 'billing_state', $state );
    }
    if ( isset( $_POST['bs_addr_city'] ) ) {
        update_user_meta( $user_id, 'billing_city', sanitize_text_field( wp_unslash( $_POST['bs_addr_city'] ) ) );
    }
    if ( isset( $_POST['bs_addr_address'] ) ) {
        update_user_meta( $user_id, 'billing_address_1', sanitize_textarea_field( wp_unslash( $_POST['bs_addr_address'] ) ) );
    }
    if ( isset( $_POST['bs_addr_postcode'] ) ) {
        update_user_meta( $user_id, 'billing_postcode', sanitize_text_field( wp_unslash( $_POST['bs_addr_postcode'] ) ) );
    }
    update_user_meta( $user_id, 'billing_country', 'IR' );
}

function bs_account_nav_build_v4( $items ) {
    $cfg = bs_account_nav_config_v4();

    $vars = function_exists( 'WC' ) && WC()->query
        ? WC()->query->get_query_vars()
        : array();

    if ( isset( $vars['study-hours'] ) ) {
        $items['study-hours'] = current_user_can( 'bs_counselor' )
            ? 'ثبت مطالعه دانش‌آموزان'
            : 'ثبت ساعت مطالعه';
    }

    if ( isset( $vars['bse-exams'] ) ) {
        $items['bse-exams'] = 'آزمون‌ها';
    }

    if ( isset( $vars['bse-results'] ) ) {
        $items['bse-results'] = 'نتایج آزمون';
    }

    if (
        isset( $vars['bse-exams'] )
        && (
            ! defined( 'BSE_KEEP_THEME_QUIZ' )
            || ! BSE_KEEP_THEME_QUIZ
        )
    ) {
        unset( $items['quiz-results'] );
    }

    unset(
        $items['edit-address'],
        $items['payment-methods'],
        $items['notifications']
    );

    $items['bs-shop'] = $cfg['shop_label'];

    $result = array();

    foreach ( $cfg['order'] as $key ) {
        if (
            'customer-logout' !== $key
            && 'bs-shop' !== $key
            && isset( $items[ $key ] )
        ) {
            $result[ $key ] = $items[ $key ];
            unset( $items[ $key ] );
        }
    }

    $logout = isset( $items['customer-logout'] )
        ? $items['customer-logout']
        : null;

    unset(
        $items['customer-logout'],
        $items['bs-shop']
    );

    $result += $items;
    $result['bs-shop'] = $cfg['shop_label'];

    if ( null !== $logout ) {
        $result['customer-logout'] = $logout;
    }

    return $result;
}
