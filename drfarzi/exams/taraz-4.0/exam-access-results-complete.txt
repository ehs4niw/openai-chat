function bse_ready() {
    return function_exists( 'bse_is_mgr' )
        && function_exists( 'bse_tbl' )
        && function_exists( 'bse_finalize' )
        && function_exists( 'bse_card' )
        && function_exists( 'bse_row' )
        && function_exists( 'bse_exam' );
}

function bse_ok() {
    return bse_ready() && bse_is_mgr();
}

function bse_cap() {
    return current_user_can( 'manage_options' )
        ? 'manage_options'
        : 'manage_woocommerce';
}

function bse_fa( $s ) {
    return strtr(
        (string) $s,
        array(
            '0' => '۰', '1' => '۱', '2' => '۲',
            '3' => '۳', '4' => '۴', '5' => '۵',
            '6' => '۶', '7' => '۷', '8' => '۸', '9' => '۹',
        )
    );
}

function bse_gregorian_to_jalali( $gy, $gm, $gd ) {
    $gdm = array( 0,31,59,90,120,151,181,212,243,273,304,334 );
    $gy2 = $gm > 2 ? $gy + 1 : $gy;
    $days = 355666 + 365 * $gy + intdiv( $gy2 + 3, 4 ) - intdiv( $gy2 + 99, 100 )
        + intdiv( $gy2 + 399, 400 ) + $gd + $gdm[$gm - 1];
    $jy = -1595 + 33 * intdiv( $days, 12053 );
    $days %= 12053;
    $jy += 4 * intdiv( $days, 1461 );
    $days %= 1461;
    if ( $days > 365 ) { $jy += intdiv( $days - 1, 365 ); $days = ( $days - 1 ) % 365; }
    if ( $days < 186 ) { $jm = 1 + intdiv( $days, 31 ); $jd = 1 + $days % 31; }
    else { $jm = 7 + intdiv( $days - 186, 30 ); $jd = 1 + ( $days - 186 ) % 30; }
    return array( $jy, $jm, $jd );
}
function bse_dt( $timestamp ) {
    if ( (int) $timestamp <= 0 ) { return '—'; }
    $date = ( new DateTimeImmutable( '@' . (int) $timestamp ) )->setTimezone( wp_timezone() );
    list( $jy, $jm, $jd ) = bse_gregorian_to_jalali( (int) $date->format('Y'), (int) $date->format('n'), (int) $date->format('j') );
    $months = array( 'فروردین','اردیبهشت','خرداد','تیر','مرداد','شهریور','مهر','آبان','آذر','دی','بهمن','اسفند' );
    return bse_fa( $jd . ' ' . $months[$jm - 1] . ' ' . $jy . '، ' . $date->format('H:i') );
}

function bse_mm( $seconds ) {
    $seconds = max( 0, (int) $seconds );
    $hours = floor( $seconds / 3600 );
    $minutes = floor( $seconds % 3600 / 60 );
    $rest = $seconds % 60;

    return bse_fa(
        ( $hours ? sprintf( '%02d:', $hours ) : '' )
        . sprintf( '%02d:%02d', $minutes, $rest )
    );
}

function bse_pct( $percent ) {
    $value = number_format( (float) $percent, 2, '.', '' );
    $value = rtrim( rtrim( $value, '0' ), '.' );

    return bse_fa(
        str_replace( '.', '٫', $value )
    ) . '٪';
}

function bse_sweep( $uid = 0 ) {
    // Account cards only finalize their own user's attempts, never everybody's results.
    if (!$uid) { return bse_close_due(); }
    global $wpdb;
    $last=0;
    do {
        $rows=$wpdb->get_results($wpdb->prepare('SELECT id,exam_id FROM '.bse_tbl().' WHERE user_id=%d AND submitted=0 AND id>%d ORDER BY id ASC LIMIT 25',$uid,$last));
        foreach($rows as $a){$last=(int)$a->id;bse_my((int)$a->exam_id,(int)$uid);}
    }while(count($rows)===25);
}

/* ---------- محصول و دسترسی ---------- */

function bse_prods( $exam_id ) {
    $products = get_post_meta(
        $exam_id,
        'bse_products',
        true
    );

    return is_array( $products )
        ? array_values(
            array_filter(
                array_map( 'intval', $products )
            )
        )
        : array();
}

function bse_days( $exam_id ) {
    $days = get_post_meta( $exam_id, 'bse_days', true );

    if ( '' === $days || false === $days ) {
        return defined( 'BSE_DEFAULT_DAYS' )
            ? max( 0, (int) BSE_DEFAULT_DAYS )
            : 7;
    }

    return max( 0, (int) $days );
}

function bse_buys( $uid ) {
    static $cache = array();

    if ( isset( $cache[ $uid ] ) ) {
        return $cache[ $uid ];
    }

    $out = array();

    if ( $uid && function_exists( 'wc_get_orders' ) ) {
        $user = get_userdata( $uid );
        $email = $user
            ? strtolower( (string) $user->user_email )
            : '';

        $base = array(
            'status' => array(
                'wc-processing',
                'wc-completed',
            ),
            'limit' => -1,
        );

        $orders = array();

        foreach (
            wc_get_orders(
                $base + array( 'customer_id' => $uid )
            ) as $order
        ) {
            $orders[ $order->get_id() ] = $order;
        }

        if ( $email ) {
            foreach (
                wc_get_orders(
                    $base + array(
                        'billing_email' => $email,
                    )
                ) as $order
            ) {
                if (
                    strtolower(
                        (string) $order->get_billing_email()
                    ) === $email
                ) {
                    $orders[ $order->get_id() ] = $order;
                }
            }
        }

        foreach ( $orders as $order ) {
            $date = $order->get_date_paid()
                ? $order->get_date_paid()
                : $order->get_date_created();

            $timestamp = $date ? $date->getTimestamp() : 0;

            foreach ( $order->get_items() as $item ) {
                foreach (
                    array(
                        (int) $item->get_product_id(),
                        (int) $item->get_variation_id(),
                    ) as $product_id
                ) {
                    if (
                        $product_id
                        && (
                            ! isset( $out[ $product_id ] )
                            || $out[ $product_id ] < $timestamp
                        )
                    ) {
                        $out[ $product_id ] = $timestamp;
                    }
                }
            }
        }
    }

    $cache[ $uid ] = $out;
    return $out;
}

function bse_acc( $exam_id, $uid ) {
    $products = bse_prods( $exam_id );

    $result = array(
        'linked' => (bool) $products,
        'ever' => false,
        'active' => false,
        'until' => 0,
        'pid' => $products ? $products[0] : 0,
    );

    if ( ! $products || ! $uid ) {
        return $result;
    }

    $purchases = bse_buys( $uid );
    $best = 0;

    foreach ( $products as $product_id ) {
        if ( isset( $purchases[ $product_id ] ) ) {
            $result['ever'] = true;

            if ( $purchases[ $product_id ] >= $best ) {
                $best = $purchases[ $product_id ];
                $result['pid'] = $product_id;
            }
        }
    }

    if ( $result['ever'] ) {
        $days = bse_days( $exam_id );

        if ( 0 === $days ) {
            $result['active'] = true;
        } else {
            $result['until'] = $best + $days * DAY_IN_SECONDS;
            $result['active'] = time() <= $result['until'];
        }
    }

    return $result;
}

function bse_can( $exam_id, $uid ) {
    if ( ! $uid ) {
        return false;
    }

    if ( bse_is_mgr() ) {
        return true;
    }

    $access = bse_acc( $exam_id, $uid );

    if ( ! $access['linked'] ) {
        return defined( 'BSE_OPEN_WITHOUT_PRODUCT' )
            && BSE_OPEN_WITHOUT_PRODUCT;
    }

    return $access['active'];
}

function bse_deny( $exam_id, $uid ) {
    $access = bse_acc( $exam_id, $uid );

    if (
        $access['linked']
        && $access['ever']
        && ! $access['active']
    ) {
        return 'دسترسی شما به این آزمون تمام شده است؛ برای شرکت دوباره باید محصول مربوط به آن را بخرید';
    }

    return 'برای شرکت در این آزمون باید محصول مربوط به آن را خریداری کنید';
}

function bse_gate_card( $post, $uid ) {
    $id = $post->ID;
    $row = bse_row( $id, $uid );
    $access = bse_acc( $id, $uid );

    if ( ! $access['linked'] ) {
        if (
            $row
            || (
                defined( 'BSE_OPEN_WITHOUT_PRODUCT' )
                && BSE_OPEN_WITHOUT_PRODUCT
            )
        ) {
            return bse_card( $post, $uid );
        }

        return null;
    }

    if ( ! $row && ! $access['ever'] ) {
        return null;
    }

    $card = bse_card( $post, $uid );

    if ( $access['active'] ) {
        if ( $access['until'] ) {
            $card['until'] = $access['until'];
        }
    } elseif ( ! $row ) {
        if ( 'ended' === $card['status'] ) {
            return null;
        }

        $card['lock'] = array(
            'name' => get_the_title( $access['pid'] ),
            'url' => get_permalink( $access['pid'] ),
        );
    }

    return $card;
}

add_action( 'wp_ajax_bse', 'bse_gate', 1 );

function bse_gate() {
    if ( ! bse_ready() || bse_is_mgr() ) {
        return;
    }

    $operation = isset( $_POST['op'] )
        ? sanitize_key( $_POST['op'] )
        : '';

    $uid = get_current_user_id();

    if ( 'list' === $operation ) {
        check_ajax_referer( 'bse', 'n' );

        $out = array();

        $posts = get_posts(
            array(
                'post_type' => 'bs_exam',
                'numberposts' => 200,
                'orderby' => 'date',
                'order' => 'DESC',
            )
        );

        foreach ( $posts as $post ) {
            $card = bse_gate_card( $post, $uid );

            if ( $card ) {
                $out[] = $card;
            }
        }

        wp_send_json_success( $out );
    }

    if (
        in_array(
            $operation,
            array(
                'get', 'start', 'save',
                'submit', 'result', 'board',
            ),
            true
        )
    ) {
        check_ajax_referer( 'bse', 'n' );

        $id = isset( $_POST['id'] )
            ? (int) $_POST['id']
            : 0;

        $post = bse_exam( $id );

        if ( ! $post ) {
            wp_send_json_error( 'آزمون پیدا نشد' );
        }

        if ( 'get' === $operation ) {
            $card = bse_gate_card( $post, $uid );

            if ( ! $card ) {
                wp_send_json_error( bse_deny( $id, $uid ) );
            }

            wp_send_json_success( $card );
        }

        if (
            ! bse_row( $id, $uid )
            && ! bse_can( $id, $uid )
        ) {
            wp_send_json_error( bse_deny( $id, $uid ) );
        }
    }
}

add_filter(
    'woocommerce_account_menu_items',
    function ( $items ) {
        $mine = array(
            'bse-exams' => 'آزمون‌ها',
            'bse-results' => 'نتایج آزمون',
        );

        foreach ( $mine as $key => $label ) {
            unset( $items[ $key ] );
        }

        if (
            ! defined( 'BSE_KEEP_THEME_QUIZ' )
            || ! BSE_KEEP_THEME_QUIZ
        ) {
            unset( $items['quiz-results'] );
        }

        $anchor = '';

        foreach (
            array(
                'study-hours',
                'my-courses',
                'dashboard',
            ) as $key
        ) {
            if ( isset( $items[ $key ] ) ) {
                $anchor = $key;
                break;
            }
        }

        if ( '' === $anchor ) {
            return $mine + $items;
        }

        $new = array();

        foreach ( $items as $key => $label ) {
            $new[ $key ] = $label;

            if ( $key === $anchor ) {
                $new = $new + $mine;
            }
        }

        return $new;
    },
    100000
);

    if (releases.length && serverNow) {

        return String(value || '')
            .replace(/[يى]/g, 'ی')
            .replace(/ك/g, 'ک')
            .replace(/[۰-۹]/g, function (digit) {
                return '۰۱۲۳۴۵۶۷۸۹'.indexOf(digit);
            })
            .replace(/[٠-٩]/g, function (digit) {
                return '٠١٢٣٤٥٦٧٨٩'.indexOf(digit);
            })
            .replace(
                /[\s\u200c\u200e\u200f\u202a-\u202e\u2066-\u2069-]+/g,
                ''
            )

            function (digit) {
                return '۰۱۲۳۴۵۶۷۸۹'.charAt(+digit);
            }
        );

            : '';

function bse_att_query( $exam, $search, $limit, $offset ) {
    global $wpdb;

    $table = bse_tbl();
    $where = 'a.submitted>0';
    $args = array();

    if ( $exam ) {
        $where .= ' AND a.exam_id=%d';
        $args[] = $exam;
    }

    if ( '' !== $search ) {
        $like = '%' . $wpdb->esc_like( $search ) . '%';

        $where .= " AND (
            u.display_name LIKE %s
            OR u.user_login LIKE %s
            OR u.user_email LIKE %s
            OR EXISTS(
                SELECT 1
                FROM {$wpdb->usermeta} m
                WHERE m.user_id=a.user_id
                AND m.meta_key IN (
                    'first_name',
                    'last_name',
                    'billing_phone'
                )
                AND m.meta_value LIKE %s
            )
        )";

        array_push(
            $args,
            $like,
            $like,
            $like,
            $like
        );
    }

    $from = "FROM $table a
        LEFT JOIN {$wpdb->users} u ON u.ID=a.user_id
        JOIN {$wpdb->posts} p ON p.ID=a.exam_id";

    $order = $exam
        ? 'a.pct DESC, a.secs ASC'
        : 'a.submitted DESC';

    $summary_query = "SELECT COUNT(*) c, AVG(a.pct) v
        $from WHERE $where";

    $summary = $wpdb->get_row(
        $args
            ? $wpdb->prepare( $summary_query, $args )
            : $summary_query
    );

    $row_args = $args;
    $row_args[] = (int) $limit;
    $row_args[] = (int) $offset;

    $rows = $wpdb->get_results(
        $wpdb->prepare(
            "SELECT a.*, p.post_title
            $from WHERE $where
            ORDER BY $order
            LIMIT %d OFFSET %d",
            $row_args
        )
    );

    return array(
        $rows,
        (int) $summary->c,
        (float) $summary->v,
    );
}

    return $user ? $user->user_login : '';
}

function bse_page_attempts() {
    if ( ! bse_ok() ) {
        return;
    }

    if(false===bse_sweep())wp_die('نتایج در حال نهایی‌سازی است؛ چند ثانیه بعد صفحه را تازه کنید.');

    $exam = isset( $_GET['exam'] )
        ? (int) $_GET['exam']
        : 0;

    $search = isset( $_GET['s'] )
        ? sanitize_text_field(
            wp_unslash( $_GET['s'] )
        )
        : '';

    $per_page = 30;

    $page = max(
        1,
        isset( $_GET['paged'] )
            ? (int) $_GET['paged']
            : 1
    );

    $offset = ( $page - 1 ) * $per_page;

    list( $rows, $total, $average ) = bse_att_query(
        $exam,
        $search,
        $per_page,
        $offset
    );

    $exams = get_posts(
        array(
            'post_type' => 'bs_exam',
            'numberposts' => 200,
            'orderby' => 'date',
            'order' => 'DESC',
        )
    );

    $pages = max(
        1,
        (int) ceil( $total / $per_page )

    if ( $pages > 1 ) {
        echo '<div class="tablenav">';
        echo '<div class="tablenav-pages">';

        echo paginate_links(
            array(
                'base' => add_query_arg( 'paged', '%#%' ),

                'current' => $page,
                'total' => $pages,
            )
        );

    }

}

add_action( 'admin_post_bse_csv', function () {
    if ( ! bse_ok() ) {
        wp_die( 'دسترسی ندارید' );
    }

    check_admin_referer( 'bse_csv' );
    if(false===bse_sweep())wp_die('نتایج در حال نهایی‌سازی است؛ چند ثانیه بعد صفحه را تازه کنید.');

    $exam = isset( $_GET['exam'] )
        ? (int) $_GET['exam']
        : 0;

    $search = isset( $_GET['s'] )
        ? sanitize_text_field(
            wp_unslash( $_GET['s'] )
        )
        : '';

    list( $rows ) = bse_att_query(
        $exam,
        $search,
        100000,
        0
    );

    $lines = array(
        array(
            'رتبه', 'نام', 'موبایل', 'آزمون',
            'درصد', 'صحیح', 'غلط', 'بدون پاسخ',
            'زمان (ثانیه)', 'تاریخ شرکت',
        ),
    );

    foreach ( $rows as $index => $row ) {
        $lines[] = array(
            $exam ? $index + 1 : '',
            bse_name( $row->user_id ),
            '="' . bse_mobile( $row->user_id ) . '"',
            $row->post_title,
            $row->pct,
            $row->correct,
            $row->wrong,
            $row->blank,
            $row->secs,
            wp_date(
                'Y-m-d H:i',
                (int) $row->submitted
            ),
        );
    }

    nocache_headers();

    header( 'Content-Type: text/csv; charset=utf-8' );

    header(
        'Content-Disposition: attachment; filename=exam-results-'
        . wp_date( 'Ymd' )
        . '.csv'
    );

    echo "\xEF\xBB\xBF";

    foreach ( $lines as $line ) {
        echo implode(
            ',',
            array_map(
                function ( $cell ) {
                    $cell = (string) $cell;

                    return (
                        preg_match( '/[",\n]/', $cell )
                        && $cell[0] !== '='
                    )
                        ? '"'
                            . str_replace(
                                '"',
                                '""',
                                $cell
                            )
                            . '"'
                        : $cell;
                },
                $line
            )
        ) . "\r\n";
    }

    exit;
} );

                    get_post_meta( $post->ID, 'en', true )

            )

                    : ' — <b style="color:#b32d2e">بدون محصول (فقط مدیر)</b>'

add_action( 'admin_post_bse_access', function () {
    if ( ! bse_ok() ) {
        wp_die( 'دسترسی ندارید' );
    }

    check_admin_referer( 'bse_access' );

    $ids = isset( $_POST['bse_e'] )
        ? array_map( 'intval', (array) $_POST['bse_e'] )
        : array();

    foreach ( $ids as $exam_id ) {
        if ( ! bse_exam( $exam_id ) ) {
            continue;
        }

        $selected = isset( $_POST['bse_p'][ $exam_id ] )
            ? array_values(
                array_unique(
                    array_filter(
                        array_map(
                            'intval',
                            (array) $_POST['bse_p'][ $exam_id ]
                        )
                    )
                )
            )
            : array();

        update_post_meta(
            $exam_id,
            'bse_products',
            $selected
        );

        if ( isset( $_POST['bse_d'][ $exam_id ] ) ) {
            update_post_meta(
                $exam_id,
                'bse_days',
                max(
                    0,
                    min(
                        3650,
                        (int) $_POST['bse_d'][ $exam_id ]
                    )
                )
            );
        }
    }

    wp_safe_redirect(
        add_query_arg(
            array(
                'page' => 'bse-access',
                'saved' => 1,
            ),
            admin_url( 'admin.php' )
        )
    );

    exit;
} );

/* ---------- اتصال محصول از پنل آزمون‌ها ---------- */

add_action(
    'wp_ajax_bse',
    function () {
        $operation = isset( $_POST['op'] )
            ? sanitize_key( $_POST['op'] )
            : '';

        if (
            ! in_array(
                $operation,
                array(
                    'access_get',
                    'access_save',
                    'access_search',
                ),
                true
            )
        ) {
            return;
        }

        check_ajax_referer( 'bse', 'n' );

        if (
            ! bse_ok()
            || ! function_exists( 'wc_get_products' )
        ) {
            wp_send_json_error(
                'دسترسی ندارید یا ووکامرس فعال نیست.'
            );
        }

        $id = isset( $_POST['id'] )
            ? absint( $_POST['id'] )
            : 0;

        if ( ! bse_exam( $id ) ) {
            wp_send_json_error( 'آزمون پیدا نشد.' );
        }

        $selected = bse_prods( $id );

        if ( 'access_save' === $operation ) {
            $raw = isset( $_POST['products'] )
                ? json_decode(
                    wp_unslash( $_POST['products'] ),
                    true
                )
                : null;

            $days = isset( $_POST['days'] )
                ? filter_var(
                    $_POST['days'],
                    FILTER_VALIDATE_INT
                )
                : false;

            if (
                ! is_array( $raw )
                || count( $raw ) > 1000
                || false === $days
                || $days < 0
                || $days > 3650
            ) {
                wp_send_json_error(
                    'محصول‌ها یا مدت دسترسی معتبر نیست.'
                );
            }

            $ids = array();

            foreach ( $raw as $product_id ) {
                if (
                    ! is_numeric( $product_id )
                    || (int) $product_id <= 0
                    || (float) $product_id
                        != (int) $product_id
                ) {
                    wp_send_json_error(
                        'شناسه محصول معتبر نیست.'
                    );
                }

                $product = wc_get_product(
                    (int) $product_id
                );

                if (
                    ! $product
                    || (
                        'publish' !== $product->get_status()
                        && ! in_array(
                            (int) $product_id,
                            $selected,
                            true
                        )
                    )
                ) {
                    wp_send_json_error(
                        'یکی از محصول‌ها قابل انتخاب نیست.'
                    );
                }

                $ids[] = (int) $product_id;
            }

            update_post_meta(
                $id,
                'bse_products',
                array_values( array_unique( $ids ) )
            );

            update_post_meta(
                $id,
                'bse_days',
                $days
            );

            wp_send_json_success(
                array( 'saved' => 1 )
            );
        }

        $term = isset( $_POST['q'] )
            ? sanitize_text_field(
                wp_unslash( $_POST['q'] )
            )
            : '';

        $page = max(
            1,
            isset( $_POST['page'] )
                ? absint( $_POST['page'] )
                : 1
        );

        $args = array(
            'status' => 'publish',
            'limit' => 30,
            'page' => $page,
            'paginate' => true,
            'orderby' => 'title',
            'order' => 'ASC',
        );

        if ( '' !== $term ) {
            $args['s'] = $term;
        }

        $found = wc_get_products( $args );
        $items = array();

        foreach ( $found->products as $product ) {
            $items[] = array(
                'id' => $product->get_id(),
                'name' => $product->get_name(),
            );
        }

        $chosen = array();

        foreach ( $selected as $product_id ) {
            $chosen[] = array(
                'id' => (int) $product_id,
                'name' => get_the_title( $product_id )
                    ?: 'محصول حذف‌شده #' . $product_id,
            );
        }

        wp_send_json_success(
            array(
                'title' => get_the_title( $id ),
                'selected' => $chosen,
                'days' => bse_days( $id ),
                'items' => $items,
                'more' => $page < $found->max_num_pages,
            )
        );
    },
    0
);
