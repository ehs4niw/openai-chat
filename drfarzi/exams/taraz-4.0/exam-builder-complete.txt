/* فایل‌های چندگانه؛ سازگار با آزمون‌های تک‌فایلی قبلی */
function bse_pdf_files( $id ) {
    $files = get_post_meta( $id, 'pdfs', true );
    if ( ! is_array( $files ) || ! $files ) {
        $old = get_post_meta( $id, 'pdf', true );
        $files = is_string( $old ) && $old !== '' ? array( $old ) : array();
    }
    return array_values( array_filter( $files, 'is_string' ) );
}

/* زمان انتشار کارنامه: پایان مهلت عمومی یا پایان زمان شخصی؛ ثبت زودهنگام آن را جلو نمی‌اندازد. */
function bse_early_results( $id ) {
    return 1 === (int) get_post_meta( $id, 'bse_early_results', true );
}
function bse_release_at( $id, $attempt = null ) {
    $end = (int) get_post_meta( $id, 'en', true );
    if ( bse_early_results( $id ) && $attempt && (int) $attempt->deadline > 0 ) {
        return $end > 0 ? min( $end, (int) $attempt->deadline ) : (int) $attempt->deadline;
    }
    return $end;
}
function bse_result_visible( $id, $attempt, $now = null ) {
    $now = null === $now ? time() : (int) $now;
    return $attempt && (int) $attempt->submitted > 0 && $now >= bse_release_at( $id, $attempt );
}
function bse_rank_is_temporary( $id, $now = null ) {
    $now = null === $now ? time() : (int) $now;
    return bse_early_results( $id ) && $now < (int) get_post_meta( $id, 'en', true );
}
/* در رتبه‌بندی موقت فقط تلاش‌هایی می‌آیند که زمان پاسخ‌گویی‌شان پایان یافته است. */
function bse_rank_scope( $id, $now = null ) {
    $now = null === $now ? time() : (int) $now;
    return bse_rank_is_temporary( $id, $now ) ? ' AND deadline<=' . $now : '';
}

}

function bse_install() {
	global $wpdb;
	if ( get_option( 'bse_db' ) !== '1' ) {
		require_once ABSPATH . 'wp-admin/includes/upgrade.php';
		$c = $wpdb->get_charset_collate();
		dbDelta( 'CREATE TABLE ' . bse_tbl() . " (
			id bigint(20) unsigned NOT NULL AUTO_INCREMENT,
			exam_id bigint(20) unsigned NOT NULL,
			user_id bigint(20) unsigned NOT NULL,
			answers longtext,
			started int(11) unsigned NOT NULL DEFAULT 0,
			deadline int(11) unsigned NOT NULL DEFAULT 0,
			submitted int(11) unsigned NOT NULL DEFAULT 0,
			correct smallint(6) NOT NULL DEFAULT 0,
			wrong smallint(6) NOT NULL DEFAULT 0,
			blank smallint(6) NOT NULL DEFAULT 0,
			pct float NOT NULL DEFAULT 0,
			secs int(11) NOT NULL DEFAULT 0,
			PRIMARY KEY  (id),
			UNIQUE KEY ux (exam_id,user_id)
		) $c;" );
		update_option( 'bse_db', '1' );
	}
	/* یک‌بار: نتیجه‌های قبلی با قانون جدید (درصد منفی) دوباره حساب می‌شوند */
	if ( get_option( 'bse_v' ) !== '2' ) {
		foreach ( $wpdb->get_results( 'SELECT * FROM ' . bse_tbl() . ' WHERE submitted>0' ) as $a ) {
			bse_finalize( $a, (int) $a->submitted );
		}
		update_option( 'bse_v', '2' );
	}
}

function bse_name( $uid ) {
	$u = get_userdata( $uid );
	if ( ! $u ) {
		return '—';
	}
	$f = get_user_meta( $uid, 'first_name', true );
	$l = get_user_meta( $uid, 'last_name', true );
	return trim( $f . ' ' . $l ) ? trim( $f . ' ' . $l ) : $u->display_name;
}

	$f = get_user_meta( $uid, 'first_name', true );
	$l = get_user_meta( $uid, 'last_name', true );
	if ( $f || $l ) {
		return trim( $f . ' ' . ( $l ? mb_substr( $l, 0, 1 ) . '.' : '' ) );
	}
	return bse_name( $uid );
}

/* فرصت کوتاه برای رسیدن بستهٔ نهایی؛ زمان پاسخ‌گویی افزایش نمی‌یابد. */
function bse_delivery_grace() { return 15; }
function bse_attempt_lock( $id ) {
 global $wpdb;
 return 1 === (int) $wpdb->get_var( $wpdb->prepare( 'SELECT GET_LOCK(%s, 5)', 'bse_' . md5( bse_tbl() . ':' . $id ) ) );
}
function bse_attempt_unlock( $id ) {
 global $wpdb;
 $wpdb->get_var( $wpdb->prepare( 'SELECT RELEASE_LOCK(%s)', 'bse_' . md5( bse_tbl() . ':' . $id ) ) );
}
function bse_finalize( $a, $t = 0, $locked = false ) {
 global $wpdb;
 $attempt_id = (int) $a->id;
 if ( ! $locked && ! bse_attempt_lock( $attempt_id ) ) { return; }
 try {
 $a = $wpdb->get_row( $wpdb->prepare( 'SELECT * FROM ' . bse_tbl() . ' WHERE id=%d', $attempt_id ) );
 if ( ! $a || ( $a->submitted && ! $t ) ) { return; }
	$id  = (int) $a->exam_id;
	$key = json_decode( get_post_meta( $id, 'key', true ), true );
	$key = is_array( $key ) ? $key : array();
	$an  = json_decode( $a->answers, true );
	$an  = is_array( $an ) ? $an : array();
	$qn  = (int) get_post_meta( $id, 'qn', true );
	$c   = 0;
	$w   = 0;
	for ( $i = 1; $i <= $qn; $i++ ) {
		$v = isset( $an[ $i ] ) ? (int) $an[ $i ] : 0;
		if ( ! $v ) {
			continue;
		}
		if ( isset( $key[ $i - 1 ] ) && (int) $key[ $i - 1 ] === $v ) {
			$c++;
		} else {
			$w++;
		}
	}
	$neg = (int) get_post_meta( $id, 'neg', true );
	/* نمره × ۳ (عدد صحیح) تا خطای اعشاری پیش نیاید؛ درصد منفی هم مجاز است و دو رقم اعشار بدون رُند می‌شود */
	$s3  = $neg ? ( 3 * $c - $w ) : ( 3 * $c );
	$raw = $qn ? $s3 * 10000 / ( 3 * $qn ) : 0;
	$pct = ( $raw < 0 ? ceil( $raw ) : floor( $raw ) ) / 100;

    if ( bsem_is( $id ) ) {
        $multi = bsem_score( $id, $an );
        $c = $multi['correct']; $w = $multi['wrong']; $pct = $multi['pct'];
    }
	$t   = $t ? $t : min( time(), (int) $a->deadline );
	$wpdb->update( bse_tbl(), array(
		'submitted' => $t,
		'correct'   => $c,
		'wrong'     => $w,
		'blank'     => $qn - $c - $w,
		'pct'       => $pct,
		'secs'      => max( 0, $t - (int) $a->started ),
	), array( 'id' => $a->id ) );
 } finally { if ( ! $locked ) { bse_attempt_unlock( $attempt_id ); } }
}
function bse_my( $e, $uid ) {
	$a = bse_row( $e, $uid );
    if ( $a && bsem_is( $e ) ) { $a = bsem_touch( $a ); }
	if ( $a && ! $a->submitted && time() >= $a->deadline + bse_delivery_grace() ) {
		bse_finalize( $a );
		$a = bse_row( $e, $uid );
	}
	return $a;
}
/* هر کس مهلتش تمام شده و ثبت نهایی نکرده، خودکار ثبت می‌شود تا در نتایج و رتبه‌ها بیاید */
/* One nonblocking sweep worker across all exam read paths; small keyset batches. */
function bse_close_due( $e = 0 ) {
 global $wpdb;
 $e=(int)$e;$table=bse_tbl();$lock='bse_sweep_'.md5($table);
 if(1!==(int)$wpdb->get_var($wpdb->prepare('SELECT GET_LOCK(%s,0)',$lock)))return false;
 $ready=true;$last=0;$began=microtime(true);$now=time();
 try {
  $multi=$e?(bsem_is($e)?array($e):array()):get_posts(array('post_type'=>'bs_exam','numberposts'=>-1,'fields'=>'ids','meta_key'=>'bsem_type','meta_value'=>'multi'));
  $multi=array_map('intval',$multi);
  do {
   $scope=$e?' AND exam_id='.$e:'';
   $eligible='deadline<='.($now-bse_delivery_grace());
   if($multi)$eligible.=' OR exam_id IN ('.implode(',',$multi).')';
   $rows=$wpdb->get_results($wpdb->prepare("SELECT id,exam_id,user_id,submitted,started,deadline FROM $table WHERE submitted=0 AND id>%d $scope AND ($eligible) ORDER BY id ASC LIMIT 25",$last));
   if($wpdb->last_error){$ready=false;break;}
   foreach($rows as $a){
    $last=(int)$a->id;
    if(bsem_is($a->exam_id)){
     $fresh=bsem_touch($a,true);
     if(!$fresh||(!$fresh->submitted && ((int)$fresh->deadline<=$now || (int)(bsem_state($fresh)['end']??PHP_INT_MAX)<=$now)))$ready=false;
    }else{
     // Never wait behind a student's active save while sweeping other attempts.
     if(1!==(int)$wpdb->get_var($wpdb->prepare('SELECT GET_LOCK(%s,0)','bse_'.md5($table.':'.$a->id)))){$ready=false;continue;}
     try{bse_finalize($a,0,true);}finally{bse_attempt_unlock($a->id);}
    }
    if(microtime(true)-$began>=1.5){$ready=false;break 2;}
   }
  }while(count($rows)===25);
  // Also wait for late final packets inside the unchanged 15-second delivery grace.
  $pending=$wpdb->get_var("SELECT COUNT(*) FROM $table WHERE submitted=0 AND deadline<=$now".($e?' AND exam_id='.$e:''));
  if($wpdb->last_error || (int)$pending>0)$ready=false;
 }finally{$wpdb->get_var($wpdb->prepare('SELECT RELEASE_LOCK(%s)',$lock));}
 return $ready;
}

/* داده‌ی کارنامه‌ی یک تلاش (برای خود شرکت‌کننده و برای مدیر) */
function bse_res( $p, $a ) {
	global $wpdb;
	$T     = bse_tbl();
	$id    = $p->ID;
    $scope = bse_rank_scope( $id );
	$rank  = 1 + (int) $wpdb->get_var( $wpdb->prepare( "SELECT COUNT(*) FROM $T WHERE exam_id=%d AND id<>%d AND submitted>0 $scope AND (pct>(SELECT pct FROM $T AS bse_self_score WHERE id=%d) OR (pct=(SELECT pct FROM $T AS bse_self_equal WHERE id=%d) AND secs<(SELECT secs FROM $T AS bse_self_time WHERE id=%d)))", $id, $a->id, $a->id, $a->id, $a->id ) );
	$total = (int) $wpdb->get_var( $wpdb->prepare( "SELECT COUNT(*) FROM $T WHERE exam_id=%d AND submitted>0 $scope", $id ) );
	$an    = json_decode( $a->answers, true );
    $scaled=bsem_is($id)?bset_cohort($id):null;$score=$scaled?($scaled['scores'][(int)$a->id]??null):null;if($scaled){$total=count($scaled['rows']);$rank=$score?$score['rank']:null;}
	return array(
        'submitted'=>(int)$a->submitted,'score'=>$score?$score['score']:null,'score_summary'=>$scaled?$scaled['summary']:null,'taraz_model'=>$score?'booklet-mean-v1':null,
		'multi' => bsem_is($id) ? 1 : 0,
        'books' => $score?$score['books']:(bsem_is($id)?bsem_score($id,is_array($an)?$an:array())['books']:array()),
        'explanation_pdf' => bsem_explanation($id),
		'pct'     => $score?$score['pct']:(float)$a->pct,
		'correct' => (int) $a->correct,
		'wrong'   => (int) $a->wrong,
		'blank'   => (int) $a->blank,
		'secs'    => (int) $a->secs,
		'rank'    => $rank,
        'temporary' => bse_rank_is_temporary( $id ) ? 1 : 0,
		'total'   => $total,
		'qn'      => (int) get_post_meta( $id, 'qn', true ),
		'title'   => $p->post_title,
		'name'    => bse_name( (int) $a->user_id ),
		'pdf'     => get_post_meta( $id, 'pdf', true ),
        'pdfs' => bse_pdf_files( $id ),
		'neg'     => (int) get_post_meta( $id, 'neg', true ),
		'mine'    => $an ? $an : new stdClass(),
		'key'     => json_decode( get_post_meta( $id, 'key', true ), true ),
	);
}
function bse_card( $p, $uid ) {
	global $wpdb;
	$id = $p->ID;
	$st = (int) get_post_meta( $id, 'st', true );
	$en = (int) get_post_meta( $id, 'en', true );
	$n  = time();
	$a  = bse_my( $id, $uid );
	$o  = array(
		'id'     => $id,
        'multi' => bsem_is($id) ? 1 : 0,
        'book_count' => bsem_is($id) ? count(bsem_books($id)) : 0,
        'practice_available' => bse_practice_available($id,$uid) ? 1 : 0,
		'title'  => $p->post_title,
		'by'     => bse_name( (int) $p->post_author ),
		'qn'     => (int) get_post_meta( $id, 'qn', true ),
		'dur'    => (int) get_post_meta( $id, 'dur', true ),
		'neg'    => (int) get_post_meta( $id, 'neg', true ),
		'st'     => $st,
		'en'     => $en,
		'now'    => $n,
		'status' => $n < $st ? 'upcoming' : ( $n < $en ? 'running' : 'ended' ),
		'me'     => $a ? ( $a->submitted ? 'done' : 'going' ) : 'none',
        'early_results' => bse_early_results( $id ) ? 1 : 0,
        'result_ready' => bse_result_visible( $id, $a, $n ) ? 1 : 0,
	);
	if ( bse_is_mgr() ) {
		$o['n'] = (int) $wpdb->get_var( $wpdb->prepare( 'SELECT COUNT(*) FROM ' . bse_tbl() . ' WHERE exam_id=%d AND submitted>0', $id ) );
	}
	return $o;
}
function bse_clean( $raw, $qn ) {
	$a = json_decode( wp_unslash( $raw ), true );
	$o = array();
	if ( is_array( $a ) ) {
		foreach ( $a as $k => $v ) {
			$k = (int) $k;
			$v = (int) $v;
			if ( $k >= 1 && $k <= $qn && $v >= 1 && $v <= 4 ) {
				$o[ $k ] = $v;
			}
		}
	}
	return wp_json_encode( (object) $o );
}

add_action( 'wp_ajax_bse', 'bse_ajax' );
function bse_ajax() {
	check_ajax_referer( 'bse', 'n' );
	global $wpdb;
	$T   = bse_tbl();
	$uid = get_current_user_id();
	$now = time();
	$op  = sanitize_key( isset( $_POST['op'] ) ? $_POST['op'] : '' );
	$id  = isset( $_POST['id'] ) ? (int) $_POST['id'] : 0;

	/* --- شرکت‌کننده --- */
	if ( $op === 'list' ) {
		if ( bse_is_mgr() ) {
			bse_close_due();
		}
		$out = array();
		foreach ( get_posts( array( 'post_type' => 'bs_exam', 'numberposts' => 100, 'orderby' => 'date', 'order' => 'DESC' ) ) as $p ) {
			$out[] = bse_card( $p, $uid );
		}
		wp_send_json_success( $out );
	}

	if ( in_array( $op, array( 'get', 'start', 'save', 'submit', 'result', 'board' ), true ) ) {
		$p = bse_exam( $id );
		if ( ! $p ) {
			wp_send_json_error( 'آزمون پیدا نشد' );
		}
		$st = (int) get_post_meta( $id, 'st', true );
		$en = (int) get_post_meta( $id, 'en', true );
		$qn = (int) get_post_meta( $id, 'qn', true );

		if ( $op === 'get' ) {
			wp_send_json_success( bse_card( $p, $uid ) );
		}

		if ( $op === 'start' ) {
            $existing = bse_my( $id, $uid );
            if ( $existing && ( $existing->submitted || $now >= $existing->deadline ) ) {
                wp_send_json_success( array( 'over' => 1 ) );
            }
			if ( $now < $st ) {
				wp_send_json_error( 'آزمون هنوز شروع نشده است' );
			}
			if ( $now >= $en ) {
				wp_send_json_error( 'زمان آزمون تمام شده است' );
			}
			$a = bse_my( $id, $uid );
			if ( ! $a ) {
				$dl = min( $now + (int) get_post_meta( $id, 'dur', true ) * 60, $en );
				$wpdb->insert( $T, array( 'exam_id' => $id, 'user_id' => $uid, 'answers' => '{}', 'started' => $now, 'deadline' => $dl ) );
				$a = bse_row( $id, $uid );
			}
			if ( $a->submitted ) {
				wp_send_json_error( 'شما قبلاً در این آزمون شرکت کرده‌اید' );
			}
			$an = json_decode( $a->answers, true );
			wp_send_json_success( array(
				'title'    => $p->post_title,
				'pdf'      => get_post_meta( $id, 'pdf', true ),
        'pdfs' => bse_pdf_files( $id ),
				'qn'       => $qn,
				'neg'      => (int) get_post_meta( $id, 'neg', true ),
				'deadline' => (int) $a->deadline,
				'now'      => $now,
				'answers'  => $an ? $an : new stdClass(),
			) );
		}

        if ( $op === 'save' || $op === 'submit' ) {
            $a = bse_row( $id, $uid );
            if ( ! $a ) { wp_send_json_error( 'آزمون شروع نشده است' ); }
            if ( ! bse_attempt_lock( $a->id ) ) { wp_send_json_error( 'ذخیره انجام نشد؛ دوباره تلاش کنید.' ); }
            $attempt_id = $a->id;
            $failed = false;
            try {
                $a = bse_row( $id, $uid );
                if ( ! $a->submitted ) {
                    // Late requests may only finalize already saved answers beyond the delivery window.
                    if ( $now <= (int) $a->deadline + bse_delivery_grace() ) {
                        if ( ! isset( $_POST['answers'] ) || ! is_array( json_decode( wp_unslash( $_POST['answers'] ), true ) ) ) {
                            $failed = true;
                        } else {
                            $js = bse_clean( $_POST['answers'], $qn );
                            $failed = $js !== $a->answers && false === $wpdb->update( $T, array( 'answers' => $js ), array( 'id' => $a->id, 'submitted' => 0 ) );
                        }
                    }
                    if ( ! $failed && ( $op === 'submit' || $now > $a->deadline + bse_delivery_grace() ) ) {
                        bse_finalize( bse_row( $id, $uid ), 0, true );
                    }
                }
                $a = bse_row( $id, $uid );
            } finally { bse_attempt_unlock( $attempt_id ); }
            if ( $failed || ( $op === 'submit' && ! $a->submitted ) ) { wp_send_json_error( 'ذخیره انجام نشد؛ دوباره تلاش کنید.' ); }
            wp_send_json_success( array( 'over' => $a->submitted ? 1 : 0 ) );
        }

        if ( $op === 'result' ) {
            $a = bse_my( $id, $uid );
            if ( ! $a ) { wp_send_json_error( 'شما در این آزمون شرکت نکرده‌اید' ); }
            $release = bse_release_at( $id, $a );
            if ( ! $a->submitted && $now >= (int) $a->deadline ) {
                wp_send_json_success( array(
                    'wait' => 1, 'pending' => 1,
                    'release' => max( $release, (int) $a->deadline + bse_delivery_grace(), $now + 3 ),
                    'now' => $now,
                ) );
            }
            if ( ! $a->submitted ) { wp_send_json_error( 'پاسخ‌های آزمون هنوز ثبت نهایی نشده است' ); }
            if ( ! bse_result_visible( $id, $a, $now ) ) {
                wp_send_json_success( array( 'wait' => 1, 'release' => $release, 'now' => $now ) );
            }
            if(!bse_close_due($id))wp_send_json_success(array('wait'=>1,'pending'=>1,'release'=>$now+3+mt_rand(0,3),'now'=>$now));
            wp_send_json_success( bse_res( $p, $a ) );
        }

        if ( $op === 'board' ) {
            if ( $now < $en ) {
                $a = bse_my( $id, $uid );
                if ( ! bse_early_results( $id ) || ( ! bse_is_mgr() && ! bse_result_visible( $id, $a, $now ) ) ) {
                    wp_send_json_error( 'نفرات برتر هنوز برای شما قابل مشاهده نیست' );
                }
            }
            if(!bse_close_due($id))wp_send_json_error('نتایج در حال نهایی‌سازی است؛ چند ثانیه بعد دوباره تلاش کنید.');
            $scope = bse_rank_scope( $id, $now );
            $rows = $wpdb->get_results( $wpdb->prepare( "SELECT user_id,pct,secs FROM $T WHERE exam_id=%d AND submitted>0 $scope ORDER BY pct DESC, secs ASC LIMIT 10", $id ) );
            $scaled=bsem_is($id)?bset_cohort($id):null;if($scaled)$rows=array_slice($scaled['rows'],0,10);
            $out = array();
            foreach ( $rows as $r ) {
                $out[] = array( 'name' => bse_short( $r->user_id ), 'pct' => (float) $r->pct, 'secs' => (int) $r->secs, 'me' => (int) $r->user_id === $uid, 'score'=>$scaled?$scaled['scores'][(int)$r->id]['score']:null, 'rank'=>$scaled?$scaled['scores'][(int)$r->id]['rank']:null );
            }
            wp_send_json_success( array( 'multi'=>bsem_is($id)?1:0,'taraz_model'=>$scaled?'booklet-mean-v1':null,'title' => $p->post_title, 'rows' => $out, 'temporary' => bse_rank_is_temporary( $id, $now ) ? 1 : 0 ) );
        }

	}

	/* --- مدیر --- */
	if ( ! bse_is_mgr() ) {
		wp_send_json_error( 'دسترسی ندارید' );
	}

	if ( $op === 'mget' ) {
		$p = bse_exam( $id );
		if ( ! $p ) {
			wp_send_json_error( 'آزمون پیدا نشد' );
		}
		wp_send_json_success( array(
			'multi' => bsem_is($id) ? 1 : 0,
            'books' => bsem_books($id),
            'explanation_pdf' => bsem_explanation($id),
            'multi_locked' => bsem_is($id) ? (int)$wpdb->get_var($wpdb->prepare('SELECT COUNT(*) FROM '.bse_tbl().' WHERE exam_id=%d',$id)) : 0,
            'early_results' => bse_early_results( $id ) ? 1 : 0,
            'title' => $p->post_title,
			'pdf'   => get_post_meta( $id, 'pdf', true ),
        'pdfs' => bse_pdf_files( $id ),
			'qn'    => (int) get_post_meta( $id, 'qn', true ),
			'dur'   => (int) get_post_meta( $id, 'dur', true ),
			'st'    => (int) get_post_meta( $id, 'st', true ),
			'en'    => (int) get_post_meta( $id, 'en', true ),
			'neg'   => (int) get_post_meta( $id, 'neg', true ),
			'key'   => json_decode( get_post_meta( $id, 'key', true ), true ),
		) );
	}

	if ( $op === 'msave' ) {
        $explanation = $id ? bsem_explanation($id) : '';
        if(isset($_POST['explanation_pdf'])){
            if(!is_string($_POST['explanation_pdf']))wp_send_json_error('فایل پاسخ‌نامهٔ تشریحی معتبر نیست.');
            $raw=trim(wp_unslash($_POST['explanation_pdf']));$explanation='';
            if($raw!==''){
                $explanation=esc_url_raw($raw,array('http','https'));$path=wp_parse_url($explanation,PHP_URL_PATH);
                if(!$explanation||!wp_parse_url($explanation,PHP_URL_HOST)||!is_string($path)||strtolower(pathinfo($path,PATHINFO_EXTENSION))!=='pdf')wp_send_json_error('پاسخ‌نامهٔ تشریحی باید PDF معتبر باشد.');
            }
        }
		$title = sanitize_text_field( wp_unslash( isset( $_POST['title'] ) ? $_POST['title'] : '' ) );
		$qn    = max( 1, min( 300, (int) $_POST['qn'] ) );
		$dur   = max( 1, (int) $_POST['dur'] );
		$st    = (int) $_POST['st'];
		$en    = (int) $_POST['en'];
		$pdf   = esc_url_raw( wp_unslash( isset( $_POST['pdf'] ) ? $_POST['pdf'] : '' ) );
		$key   = json_decode( wp_unslash( isset( $_POST['key'] ) ? $_POST['key'] : '[]' ), true );

            if ( ! is_string( $_POST['pdfs'] ) ) {
                wp_send_json_error( 'فهرست فایل‌ها معتبر نیست' );
            }
            $raw_files = json_decode( wp_unslash( $_POST['pdfs'] ), true );
            if ( ! is_array( $raw_files ) || array_values( $raw_files ) !== $raw_files ) {
                wp_send_json_error( 'فهرست فایل‌ها معتبر نیست' );
            }
            foreach ( $raw_files as $file_url ) {
                if ( ! is_string( $file_url ) ) {
                    wp_send_json_error( 'نشانی فایل معتبر نیست' );
                }
                $clean_url = esc_url_raw( $file_url, array( 'http', 'https' ) );
                $path = wp_parse_url( $clean_url, PHP_URL_PATH );
                if ( ! $clean_url || ! wp_parse_url( $clean_url, PHP_URL_HOST ) || ! is_string( $path ) || strtolower( pathinfo( $path, PATHINFO_EXTENSION ) ) !== 'pdf' ) {
                    wp_send_json_error( 'فقط فایل PDF مجاز است' );
                }
                if ( ! in_array( $clean_url, $pdfs, true ) ) {
                    $pdfs[] = $clean_url;
                }
            }
        } elseif ( $pdf ) {
            /* کلاینت قدیمی، فایل‌های اضافی آزمون را حذف نمی‌کند. */
            $pdfs = $id ? bse_pdf_files( $id ) : array();
            if ( ! $pdfs || $pdfs[0] !== $pdf ) {
                $pdfs = array( $pdf );
            }
        }
        $pdf = $pdfs ? $pdfs[0] : '';

		if ( ! $title || ! $pdf || $st <= 0 || $en <= $st || ! is_array( $key ) || count( $key ) !== $qn ) {
			wp_send_json_error( 'اطلاعات آزمون کامل یا درست نیست' );
		}
		$key = array_map( 'intval', $key );
		if ( $id ) {
			if ( ! bse_exam( $id ) ) {
				wp_send_json_error( 'آزمون پیدا نشد' );
			}
			wp_update_post( array( 'ID' => $id, 'post_title' => $title ) );
		} else {
			$id = wp_insert_post( array( 'post_type' => 'bs_exam', 'post_status' => 'publish', 'post_title' => $title ) );
		}
		update_post_meta( $id, 'pdf', $pdf );
        update_post_meta( $id, 'pdfs', $pdfs );
		update_post_meta( $id, 'qn', $qn );
		update_post_meta( $id, 'dur', $dur );
		update_post_meta( $id, 'st', $st );
		update_post_meta( $id, 'en', $en );
		update_post_meta( $id, 'neg', empty( $_POST['neg'] ) ? 0 : 1 );
        // Older clients do not erase a saved setting when the field is missing.
        if ( isset( $_POST['early_results'] ) ) {
            update_post_meta( $id, 'bse_early_results', (string) $_POST['early_results'] === '1' ? 1 : 0 );
        }
		update_post_meta( $id, 'key', wp_json_encode( $key ) );
        update_post_meta($id,'bsem_explanation',$explanation);
		/* اگر کلید عوض شده، نتیجه‌های ثبت‌شده دوباره محاسبه می‌شود */
		foreach ( $wpdb->get_results( $wpdb->prepare( "SELECT * FROM $T WHERE exam_id=%d AND submitted>0", $id ) ) as $a ) {
			bse_finalize( $a, (int) $a->submitted );
		}
		wp_send_json_success( array( 'id' => $id ) );
	}

	if ( $op === 'mres' ) {
		$p = bse_exam( $id );
		if ( ! $p ) {
			wp_send_json_error( 'آزمون پیدا نشد' );
		}
		if(!bse_close_due($id))wp_send_json_error('نتایج در حال نهایی‌سازی است؛ چند ثانیه بعد دوباره تلاش کنید.');

			wp_send_json_error( 'نتیجه‌ای برای این کاربر ثبت نشده است' );
		}
		wp_send_json_success( bse_res( $p, $a ) );
	}

	if ( $op === 'mdel' ) {
		if ( bse_exam( $id ) ) {
			wp_delete_post( $id, true );
			$wpdb->delete( $T, array( 'exam_id' => $id ) );
		}
		wp_send_json_success( 1 );
	}

	if ( $op === 'mstats' ) {
		$p = bse_exam( $id );
		if ( ! $p ) {
			wp_send_json_error( 'آزمون پیدا نشد' );
		}
		if(!bse_close_due($id))wp_send_json_error('نتایج در حال نهایی‌سازی است؛ چند ثانیه بعد دوباره تلاش کنید.');
		$qn   = (int) get_post_meta( $id, 'qn', true );
		$key  = json_decode( get_post_meta( $id, 'key', true ), true );
		$rows = $wpdb->get_results( $wpdb->prepare( "SELECT * FROM $T WHERE exam_id=%d AND submitted>0 ORDER BY pct DESC, secs ASC", $id ) );
        $scaled=bsem_is($id)?bset_cohort($id):null;if($scaled)$rows=$scaled['rows'];
		$going = (int) $wpdb->get_var( $wpdb->prepare( "SELECT COUNT(*) FROM $T WHERE exam_id=%d AND submitted=0", $id ) );
		$dist = array_fill( 0, 10, 0 );
		$qs   = array_fill( 0, $qn, 0 );
		$list = array();
		$sum  = 0;
		$r    = 0;
		foreach ( $rows as $x ) {
			$r++;
			$sum += $x->pct;
			$dist[ max( 0, min( 9, (int) floor( $x->pct / 10 ) ) ) ]++;
			$an = json_decode( $x->answers, true );
			$an = is_array( $an ) ? $an : array();
			for ( $i = 1; $i <= $qn; $i++ ) {
				if ( isset( $an[ $i ], $key[ $i - 1 ] ) && (int) $an[ $i ] === (int) $key[ $i - 1 ] ) {
					$qs[ $i - 1 ]++;
				}
			}
			$u      = get_userdata( $x->user_id );
			$mobile = get_user_meta( $x->user_id, 'billing_phone', true );
			$list[] = array(
				'rank'   => $scaled?$scaled['scores'][(int)$x->id]['rank']:$r,
                'score'=>$scaled?$scaled['scores'][(int)$x->id]['score']:null,
                'books'=>$scaled?$scaled['scores'][(int)$x->id]['books']:array(),
				'uid'    => (int) $x->user_id,
				'name'   => bse_name( $x->user_id ),
				'mobile' => $mobile ? $mobile : ( $u ? $u->user_login : '' ),
				'pct'    => (float) $x->pct,
				'c'      => (int) $x->correct,
				'w'      => (int) $x->wrong,
				'b'      => (int) $x->blank,
				'secs'   => (int) $x->secs,
			);
		}
		$n = count( $rows );
		foreach ( $qs as $i => $v ) {
			$qs[ $i ] = $n ? round( $v / $n * 100 ) : 0;
		}
		wp_send_json_success( array(
			'title' => $p->post_title,
			'pdf'   => get_post_meta( $id, 'pdf', true ),
        'pdfs' => bse_pdf_files( $id ),
			'qn'    => $qn,
			'dur'   => (int) get_post_meta( $id, 'dur', true ),
			'st'    => (int) get_post_meta( $id, 'st', true ),
			'en'    => (int) get_post_meta( $id, 'en', true ),
			'neg'   => (int) get_post_meta( $id, 'neg', true ),
			'now'   => $now,
			'n'     => $n,
			'multi' => bsem_is($id) ? 1 : 0,
            'books' => bsem_is($id) ? bsem_stats($id,$rows) : array(),
			'taraz_model'=>$scaled?'booklet-mean-v1':null,
            'score_summary'=>$scaled?$scaled['summary']:null,
            'temporary'=>bse_rank_is_temporary($id)?1:0,
			'going' => $going,
			'avg'   => $n ? round( $sum / $n, 2 ) : 0,
			'dist'  => $dist,
			'qs'    => $qs,
			'rows'  => $list,
		) );
	}

	wp_send_json_error( 'درخواست نامعتبر' );
}

function bsem_is($id) { return 'multi'===get_post_meta($id,'bsem_type',true); }

function bsem_trunc($v) { $v=round($v*100,9); return ($v<0?ceil($v):floor($v))/100; }
function bsem_score($id,$answers) {
 $key=json_decode(get_post_meta($id,'key',true),true); $key=is_array($key)?$key:array();
 $neg=(int)get_post_meta($id,'neg',true); $q=1;$total=0;$total_weights=0;$c=0;$w=0;$n=0;$out=array();
 foreach(bsem_books($id) as $b) {
  $subs=array();$sum=0;$weights=0;$bc=0;$bw=0;$bn=0;$from=$q;
  foreach($b['subjects'] as $s) {
   $sc=0;$sw=0;$sn=(int)$s['count'];$first=$q;
   for($j=0;$j<$sn;$j++,$q++) { $v=isset($answers[$q])?(int)$answers[$q]:0; if(!$v)continue; if(isset($key[$q-1])&&$v===(int)$key[$q-1])$sc++;else $sw++; }
   $pct=$sn?100*(3*$sc-($neg?$sw:0))/(3*$sn):0;
   $sum+=$pct*$s['weight'];$weights+=$s['weight'];$bc+=$sc;$bw+=$sw;$bn+=$sn;
   $subs[]=array_merge($s,array('from'=>$first,'to'=>$q-1,'pct'=>bsem_trunc($pct),'pct_raw'=>$pct,'correct'=>$sc,'wrong'=>$sw,'blank'=>$sn-$sc-$sw));
  }
  $bp=$weights?$sum/$weights:0;$total+=$bp;$total_weights++;$c+=$bc;$w+=$bw;$n+=$bn;
  $out[]=array('title'=>$b['title'],'duration'=>$b['duration'],'pdfs'=>$b['pdfs'],'from'=>$from,'to'=>$q-1,'pct'=>bsem_trunc($bp),'pct_raw'=>$bp,'correct'=>$bc,'wrong'=>$bw,'blank'=>$bn-$bc-$bw,'subjects'=>$subs);
 }
 return array('pct'=>bsem_trunc($total_weights?$total/$total_weights:0),'correct'=>$c,'wrong'=>$w,'blank'=>$n-$c-$w,'books'=>$out);
}
function bsem_state($a) {
 global $wpdb; $raw=$wpdb->get_var($wpdb->prepare('SELECT state FROM '.bsem_table().' WHERE attempt_id=%d',$a->id));
 $s=json_decode((string)$raw,true); if(is_array($s))return $s;
 $b=bsem_books($a->exam_id);return array('index'=>0,'started'=>(int)$a->started,'end'=>min((int)$a->started+$b[0]['duration']*60,(int)$a->deadline),'revision'=>0);
}
function bsem_put($a,$s) { global $wpdb; return false!==$wpdb->replace(bsem_table(),array('attempt_id'=>$a->id,'state'=>wp_json_encode($s))); }
/* Timeout consumes elapsed booklets even when the browser was closed. Caller holds attempt lock. */
function bsem_sync($a,$s,$now) {
 $books=bsem_books($a->exam_id);
 if (!$a->submitted && $s['index']>=count($books)) {
  bse_finalize($a,(int)($s['completed']??$s['end']),true);
  $a=bse_row($a->exam_id,$a->user_id);
  if(!$a->submitted)throw new RuntimeException('finalize');
 }
 while(!$a->submitted && $s['index']<count($books) && $now>=$s['end']) {
  $end=$s['end'];$s['index']++;$s['revision']++;
  if($s['index']>=count($books)) { $s['completed']=$end; if(!bsem_put($a,$s))throw new RuntimeException('state'); bse_finalize($a,$end,true);$a=bse_row($a->exam_id,$a->user_id);if(!$a->submitted)throw new RuntimeException('finalize');break; }
  $s['started']=$end;$s['end']=min($end+$books[$s['index']]['duration']*60,(int)$a->deadline);
  if(!bsem_put($a,$s))throw new RuntimeException('state');
 }
 return array($a,$s);
}
function bsem_touch($a,$background=false) {
 if(!$a||$a->submitted||!bsem_is($a->exam_id))return $a;
 $state=bsem_state($a);$books=bsem_books($a->exam_id);$now=time();
 if(isset($state['index'],$state['end']) && $state['index']<count($books) && $now<(int)$state['end'] && $now<(int)$a->deadline)return $a;
 if($background){global $wpdb;if(1!==(int)$wpdb->get_var($wpdb->prepare('SELECT GET_LOCK(%s,0)','bse_'.md5(bse_tbl().':'.$a->id))))return $a;}
 else if(!bse_attempt_lock($a->id))return $a;
 try { $a=bse_row($a->exam_id,$a->user_id); list($a,$s)=bsem_sync($a,bsem_state($a),time()); }
 catch(RuntimeException $e) { /* Leave the saved attempt available for retry. */ }
 finally { bse_attempt_unlock($a->id); }
 return $a;
}
function bsem_payload($a,$s) {
 $books=bsem_books($a->exam_id);$i=$s['index'];if($a->submitted||!isset($books[$i]))return array('over'=>1);
 $from=1;for($j=0;$j<$i;$j++)foreach($books[$j]['subjects'] as $sub)$from+=$sub['count'];

function bsem_validate($raw) {
 if(!is_array($raw)||!$raw||count($raw)>20)throw new InvalidArgumentException('بین ۱ تا ۲۰ دفترچه تعریف کنید.');
 $out=array();$qn=0;$dur=0;
 foreach($raw as $b) {
  if(!is_array($b)||!is_string($b['title']??null)||!is_array($b['pdfs']??null))throw new InvalidArgumentException('دفترچه معتبر نیست.');
  $title=sanitize_text_field($b['title']??'');$duration=filter_var($b['duration']??0,FILTER_VALIDATE_INT);
  if(!$title||!$duration||$duration<1||$duration>1440)throw new InvalidArgumentException('عنوان و مدت دفترچه را کامل کنید.');
  $pdfs=array();foreach(($b['pdfs']??array()) as $url) { if(!is_string($url))throw new InvalidArgumentException('فایل معتبر نیست.');$url=esc_url_raw($url,array('http','https'));$path=wp_parse_url($url,PHP_URL_PATH);if(!wp_parse_url($url,PHP_URL_HOST)||!is_string($path)||strtolower(pathinfo($path,PATHINFO_EXTENSION))!=='pdf')throw new InvalidArgumentException('فقط PDF معتبر مجاز است.');$pdfs[]=$url; }
  if(!$pdfs||count($pdfs)>20)throw new InvalidArgumentException('برای هر دفترچه حداقل یک PDF انتخاب کنید.');
  $subjects=array();if(empty($b['subjects'])||!is_array($b['subjects'])||count($b['subjects'])>50)throw new InvalidArgumentException('درس‌های دفترچه را تعریف کنید.');
  foreach($b['subjects'] as $s) {

   if(!$name||!$count||$count<1||$count>1000||!is_numeric($weight)||!is_finite((float)$weight)||(float)$weight<=0||(float)$weight>1000)throw new InvalidArgumentException('نام درس، تعداد سؤال صحیح و ضریب مثبت تا ۱۰۰۰ لازم است.');

function bsep_advance($s,$now) {
 if($s['submitted'])return $s;
 if($s['multi']) {
  while($s['index']<count($s['books']) && $now>=$s['book_end']) {
   $end=$s['book_end'];$s['index']++;
   if($s['index']>=count($s['books'])){$s['submitted']=$end;break;}
   $s['book_end']=$end+$s['books'][$s['index']]['duration']*60;
  }
 }elseif($now>=$s['deadline'])$s['submitted']=$s['deadline'];

}
function bsep_score($s) {
 $c=0;$w=0;$q=1;$books=array();$total=0;$total_weights=0;
 if(!$s['multi']) {
  for($q=1;$q<=$s['qn'];$q++){$v=$s['answers'][$q]??0;if(!$v)continue;if($v===(int)($s['key'][$q-1]??0))$c++;else $w++;}
  $s3=$s['neg']?3*$c-$w:3*$c;$raw=$s['qn']?$s3*10000/(3*$s['qn']):0;$pct=($raw<0?ceil($raw):floor($raw))/100;
 }else{
  foreach($s['books'] as $b){$from=$q;$subs=array();$sum=0;$weights=0;$bc=0;$bw=0;$bn=0;
   foreach($b['subjects'] as $sub){$sc=0;$sw=0;$sn=$sub['count'];$first=$q;
    for($j=0;$j<$sn;$j++,$q++){$v=$s['answers'][$q]??0;if(!$v)continue;if($v===(int)($s['key'][$q-1]??0))$sc++;else $sw++;}
    $p=$sn?100*(3*$sc-($s['neg']?$sw:0))/(3*$sn):0;$sum+=$p*$sub['weight'];$weights+=$sub['weight'];$bc+=$sc;$bw+=$sw;$bn+=$sn;
    $subs[]=array_merge($sub,array('from'=>$first,'to'=>$q-1,'pct'=>bsem_trunc($p),'correct'=>$sc,'wrong'=>$sw,'blank'=>$sn-$sc-$sw));
   }
   $bp=$weights?$sum/$weights:0;$total+=$bp;$total_weights++;$c+=$bc;$w+=$bw;
   $books[]=array_merge($b,array('from'=>$from,'to'=>$q-1,'pct'=>bsem_trunc($bp),'pct_raw'=>$bp,'correct'=>$bc,'wrong'=>$bw,'blank'=>$bn-$bc-$bw,'subjects'=>$subs));
  }
  $pct=bsem_trunc($total_weights?$total/$total_weights:0);
 }
 return array('explanation_pdf'=>$s['explanation_pdf']??'','practice'=>!empty($s['practice'])?1:0,'preview'=>empty($s['practice'])?1:0,'multi'=>$s['multi'],'title'=>$s['title'],'name'=>bse_name($s['uid']),'qn'=>$s['qn'],'pct'=>$pct,'correct'=>$c,'wrong'=>$w,'blank'=>$s['qn']-$c-$w,'secs'=>max(0,$s['submitted']-$s['started']),'rank'=>null,'total'=>0,'temporary'=>0,'pdf'=>$s['pdfs'][0]??'','pdfs'=>$s['pdfs'],'neg'=>$s['neg'],'mine'=>(object)$s['answers'],'key'=>$s['key'],'books'=>$books,'now'=>time());
}

/* Relative multi-subject scores: official final attempts only, no stored grade migration. */
function bset_calculate($id,$rows){
 $scores=array();$moments=array();$values=array();$n=count($rows);
 foreach($rows as $a){$an=json_decode($a->answers,true);$sc=bsem_score($id,is_array($an)?$an:array());$scores[(int)$a->id]=$sc;foreach($sc['books'] as $bi=>$b)foreach($b['subjects'] as $si=>$s){$k=$bi.':'.$si;$v=(float)($s['pct_raw']??$s['pct']);if(!isset($moments[$k]))$moments[$k]=array(0,0.0,0.0);$m=&$moments[$k];$m[0]++;$delta=$v-$m[1];$m[1]+=$delta/$m[0];$m[2]+=$delta*($v-$m[1]);$values[$k][]=$v;unset($m);}}
 $subjectRanks=array();foreach($values as $k=>&$v){rsort($v,SORT_NUMERIC);$last=null;$rank=0;foreach($v as $i=>$percent){if($last===null||abs($last-$percent)>1e-9)$rank=$i+1;$subjectRanks[$k][sprintf('%.9F',$percent)]=$rank;$last=$percent;}}unset($v);
 foreach($scores as &$sc){$total=0;$bp=0;$bc=0;foreach($sc['books'] as $bi=>&$b){$sum=0;$weights=0;foreach($b['subjects'] as $si=>&$s){$k=$bi.':'.$si;$v=(float)($s['pct_raw']??$s['pct']);$m=$moments[$k];$sd=sqrt(max(0,$m[2]/$m[0]));$t=$sd>1e-9?5000+1000*($v-$m[1])/$sd:5000;$t=max(3000,min(9000,$t));$s['score']=(int)round($t);$s['rank']=$subjectRanks[$k][sprintf('%.9F',$v)];$sum+=$t*(float)$s['weight'];$weights+=(float)$s['weight'];}unset($s);$b['score_raw']=$weights?$sum/$weights:5000;$b['score']=(int)round($b['score_raw']);$total+=$b['score_raw'];$bp+=(float)($b['pct_raw']??$b['pct']);$bc++;}unset($b);$sc['score_raw']=round($bc?$total/$bc:5000,6);$sc['score']=(int)round($sc['score_raw']);$sc['pct']=bsem_trunc($bc?$bp/$bc:0);}unset($sc);
 $ordered=array();foreach($rows as $a){$r=clone $a;$r->pct=$scores[(int)$r->id]['pct'];$ordered[]=$r;}
 usort($ordered,function($a,$b)use($scores){$delta=$scores[(int)$b->id]['score_raw']<=>$scores[(int)$a->id]['score_raw'];return $delta?:((int)$a->secs<=>(int)$b->secs)?:((int)$a->id<=>(int)$b->id);});
 $last=null;$rank=0;foreach($ordered as $i=>$a){$t=$scores[(int)$a->id]['score_raw'];if($last===null||$t!==$last)$rank=$i+1;$scores[(int)$a->id]['rank']=$rank;$last=$t;}
 // Sort each booklet once rather than rescanning the cohort for every person.
 $bookRanks=array();if($scores){$first=reset($scores);foreach($first['books'] as $bi=>$ignored){$values=array();foreach($scores as $sc)$values[]=round($sc['books'][$bi]['score_raw'],6);rsort($values,SORT_NUMERIC);$last=null;$rank=0;foreach($values as $i=>$v){if($last===null||$v!==$last)$rank=$i+1;$bookRanks[$bi][sprintf('%.6F',$v)]=$rank;$last=$v;}}}
 foreach($scores as &$sc)foreach($sc['books'] as $bi=>&$b)$b['rank']=$bookRanks[$bi][sprintf('%.6F',$b['score_raw'])];unset($sc,$b);
 $pcts=array_map(function($r){return (float)$r->pct;},$ordered);return array('scores'=>$scores,'rows'=>$ordered,'summary'=>array('n'=>$n,'avg'=>$n?array_sum($pcts)/$n:0,'max'=>$n?max($pcts):0,'min'=>$n?min($pcts):0));
}
function bset_cohort($id){
 static $cache=array();$scope=bse_rank_scope($id);$key=$id.'|'.$scope;if(isset($cache[$key]))return $cache[$key];global $wpdb;$rows=$wpdb->get_results($wpdb->prepare('SELECT * FROM '.bse_tbl()." WHERE exam_id=%d AND submitted>0 $scope ORDER BY id ASC",$id));if($wpdb->last_error)throw new RuntimeException('محاسبه تراز فعلاً ممکن نیست؛ دوباره تلاش کنید.');return $cache[$key]=bset_calculate($id,$rows);
}
